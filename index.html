// ============================================================ // Quiz System — Google Apps Script (Code.gs) // แก้ไข: ใช้ GET + query string ทุก action (หลีกเลี่ยง CORS) // ============================================================ const SHEET_ID = ''; // ← ใส่ Google Sheet ID ของคุณ const SHEET_QUIZ = 'QuizConfig'; const SHEET_QUESTIONS = 'Questions'; const SHEET_RESPONSES = 'Responses'; const SHEET_STAFF = 'Staff'; // ============================================================ // ENTRY POINT — ทุก action ใช้ doGet เท่านั้น (GET request) // ============================================================ // รับ POST จาก Cloudflare Worker (payload ใหญ่) function doPost(e) { try { let action = '', payload = '', name = ''; // รับ form-encoded body if (e.postData && e.postData.contents) { const body = e.postData.contents; body.split('&').forEach(function(part) { const idx = part.indexOf('='); if(idx < 0) return; const k = decodeURIComponent(part.slice(0, idx)); const v = decodeURIComponent(part.slice(idx + 1)); if(k === 'action') action = v; if(k === 'payload') payload = v; if(k === 'name') name = v; }); } const p = { action, payload, name }; let result; if (action === 'saveQuiz') result = saveQuiz(p); else if (action === 'saveResponse') result = saveResponse(p); else if (action === 'saveStaff') result = saveStaff(p); else return respond({ ok: false, error: 'Unknown action: ' + action }, ''); return respond({ ok: true, ...result }, ''); } catch(ex) { return respond({ ok: false, error: ex.message }, ''); } } function doGet(e) { const p = e.parameter; const action = p.action || ''; const cb = p.callback || ''; // JSONP callback name try { let result; if (action === 'saveQuiz') result = saveQuiz(p); else if (action === 'saveResponse') result = saveResponse(p); else if (action === 'getResults') result = getResults(); else if (action === 'getQuiz') result = getQuiz(); else if (action === 'checkName') result = checkName(p.name); else if (action === 'saveStaff') result = saveStaff(p); else if (action === 'getStaff') result = getStaff(); else return respond({ ok: false, error: 'Unknown action: ' + action }, cb); return respond({ ok: true, ...result }, cb); } catch(ex) { return respond({ ok: false, error: ex.message }, cb); } } function respond(obj, callback) { const json = JSON.stringify(obj); // ถ้ามี callback parameter ให้ตอบแบบ JSONP (browser script-tag) if (callback) { return ContentService .createTextOutput(callback + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } // ============================================================ // SAVE QUIZ CONFIG + QUESTIONS // รับ: action=saveQuiz&payload= // ============================================================ function saveQuiz(p) { let quiz; try { let raw = p.payload || ''; for(let i=0; i<4; i++){ try { const parsed = JSON.parse(raw); quiz = parsed; break; } catch(e){} try { raw = decodeURIComponent(raw); } catch(e){ break; } } if(!quiz) quiz = JSON.parse(raw); } catch(e) { return { ok: false, error: 'parse quiz payload ไม่ได้: ' + e.message }; } const ss = SpreadsheetApp.openById(SHEET_ID); // QuizConfig sheet let cfgSheet = ss.getSheetByName(SHEET_QUIZ) || ss.insertSheet(SHEET_QUIZ); cfgSheet.clearContents(); cfgSheet.appendRow(['key', 'value']); cfgSheet.appendRow(['title', quiz.title]); cfgSheet.appendRow(['expiry', String(quiz.expiry||'')]); cfgSheet.appendRow(['timer', quiz.timer]); cfgSheet.appendRow(['updated', new Date().toLocaleString('th-TH')]); styleHeader(cfgSheet); // Questions sheet let qSheet = ss.getSheetByName(SHEET_QUESTIONS) || ss.insertSheet(SHEET_QUESTIONS); qSheet.clearContents(); qSheet.appendRow(['#','คำถาม','ตัวเลือก A','ตัวเลือก B','ตัวเลือก C','ตัวเลือก D','ตัวเลือก E','ตัวเลือก F','เฉลย (index)','เฉลย (ข้อความ)','มีรูป']); styleHeader(qSheet); quiz.questions.forEach((q, i) => { qSheet.appendRow([ i+1, q.text, q.options[0]||'', q.options[1]||'', q.options[2]||'', q.options[3]||'', q.options[4]||'', q.options[5]||'', q.correct, q.options[q.correct]||'', q.hasImg ? 'มี' : '-' ]); }); return { message: 'บันทึกสำเร็จ', questions: quiz.questions.length }; } // ============================================================ // SAVE RESPONSE (คำตอบผู้เล่น) // รับ: action=saveResponse&payload= // ============================================================ function saveResponse(p) { let data; try { let raw = p.payload || ''; for(let i=0; i<4; i++){ try { const parsed = JSON.parse(raw); data = parsed; break; } catch(e){} try { raw = decodeURIComponent(raw); } catch(e){ break; } } if(!data) data = JSON.parse(raw); } catch(e) { return { ok: false, error: 'parse payload ไม่ได้: ' + e.message + ' raw=' + String(p.payload).slice(0,80) }; } // Force parse เป็น number data.correctCount = parseInt(data.correctCount) || 0; data.totalQ = parseInt(data.totalQ) || 0; const ss = SpreadsheetApp.openById(SHEET_ID); let rSheet = ss.getSheetByName(SHEET_RESPONSES); const needHeader = !rSheet || rSheet.getLastRow() === 0; if (!rSheet) rSheet = ss.insertSheet(SHEET_RESPONSES); if (needHeader) { const header = ['วันเวลา','ชื่อ','ผล','คะแนน']; (data.answers || []).forEach((_,i) => header.push('ข้อ '+(i+1))); rSheet.appendRow(header); styleHeader(rSheet); } // ตรวจชื่อซ้ำ if (rSheet.getLastRow() >= 2) { const names = rSheet.getRange(2, 2, rSheet.getLastRow()-1, 1).getValues().map(r => String(r[0]).trim().toLowerCase()); if (names.indexOf(String(data.name).trim().toLowerCase()) > -1) { return { message: 'ชื่อนี้ทำแบบทดสอบไปแล้ว', duplicate: true }; } } const row = [ data.timestamp || new Date().toLocaleString('th-TH'), data.name, data.correctCount === data.totalQ ? '✅ ถูกทั้งหมด' : '❌ ไม่ครบ', data.correctCount + '/' + data.totalQ, ...(data.answers || []).map(a => a === 1 ? '✅' : '❌') ]; rSheet.appendRow(row); const lastRow = rSheet.getLastRow(); const bg = data.correctCount === data.totalQ ? '#d1fae5' : '#fee2e2'; rSheet.getRange(lastRow, 1, 1, rSheet.getLastColumn()).setBackground(bg); try { rSheet.autoResizeColumns(1, rSheet.getLastColumn()); } catch(e){} return { message: 'บันทึกสำเร็จ', name: data.name }; } // ============================================================ // GET QUIZ (คำถาม + config สำหรับผู้เล่น) // ============================================================ function getQuiz() { const ss = SpreadsheetApp.openById(SHEET_ID); var config = {}; var cfgSheet = ss.getSheetByName(SHEET_QUIZ); if (cfgSheet) { cfgSheet.getDataRange().getValues().slice(1).forEach(function(r){ if(r[0]){ var val = r[1]; // ถ้า Sheets แปลง expiry เป็น Date object ให้แปลงกลับเป็น local string UTC+7 if(val instanceof Date){ var local = new Date(val.getTime() + 7*60*60000); val = local.toISOString().slice(0,16); } config[r[0]] = val; } }); } var qSheet = ss.getSheetByName(SHEET_QUESTIONS); if (!qSheet || qSheet.getLastRow() < 2) return { questions: [], config: config }; function cleanText(s){ return String(s||'').replace(/\+/g,' ').replace(/\s+/g,' ').trim(); } var questions = qSheet.getDataRange().getValues().slice(1).filter(function(r){ return r[1]; }).map(function(r){ return { text: cleanText(r[1]), options: [r[2],r[3],r[4],r[5],r[6],r[7]].filter(function(o){ return o !== '' && o !== null && o !== undefined; }).map(cleanText), correct: Number(r[8]) }; }); return { questions: questions, config: config }; } // ============================================================ // GET RESULTS // ============================================================ function getResults() { const ss = SpreadsheetApp.openById(SHEET_ID); const rSheet = ss.getSheetByName(SHEET_RESPONSES); if (!rSheet || rSheet.getLastRow() < 2) return { players: [] }; const rows = rSheet.getDataRange().getValues().slice(1); const players = rows.map(r => { // columns: 0=เวลา 1=ชื่อ 2=สาขา 3=แผนก 4=ตำแหน่ง 5=รหัส 6=ผล 7=คะแนน 8+=คำตอบ const scoreStr = String(r[7]||r[3]||'0/0'); const parts = scoreStr.split('/'); const correctCount = parseInt(parts[0]) || 0; const totalQ = parseInt(parts[1]) || 0; const answers = []; for (let i = 8; i < r.length; i++) { if (r[i] === '✅' || r[i] === '❌') answers.push({ correct: r[i] === '✅' }); } // fallback ถ้า column เก่า (ไม่มี branch/dept) if(!answers.length){ for (let i = 4; i < r.length; i++) { if (r[i] === '✅' || r[i] === '❌') answers.push({ correct: r[i] === '✅' }); } } const realTotal = totalQ || answers.length || 1; const realCorrect = totalQ ? correctCount : answers.filter(a=>a.correct).length; return { name:r[1], branch:String(r[2]||''), dept:String(r[3]||''), position:String(r[4]||''), empId:String(r[5]||''), correctCount:realCorrect, totalQ:realTotal, answers, timestamp:Date.now() }; }); return { players }; } // ============================================================ // STAFF — บันทึก/โหลดรายชื่อพนักงาน // ============================================================ function saveStaff(p) { let staff; try { let raw = p.payload || ''; for(let i=0; i<4; i++){ try { const parsed = JSON.parse(raw); staff = parsed; break; } catch(e){} try { raw = decodeURIComponent(raw); } catch(e){ break; } } if(!staff) staff = JSON.parse(raw); if(!Array.isArray(staff)) return { ok:false, error:'payload ไม่ใช่ array' }; } catch(e) { return { ok:false, error:'parse ไม่ได้: '+e.message }; } const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_STAFF) || ss.insertSheet(SHEET_STAFF); sheet.clearContents(); sheet.appendRow(['รหัส','ชื่อ-นามสกุล','แผนก','ตำแหน่ง','สาขา']); styleHeader(sheet); staff.forEach(function(s, i){ var name = typeof s==='object' ? (s.name||'') : String(s); var id = typeof s==='object' ? (s.id||'') : ''; var dept = typeof s==='object' ? (s.dept||'') : ''; var pos = typeof s==='object' ? (s.position||'') : ''; var br = typeof s==='object' ? (s.branch||'') : ''; sheet.appendRow([id, name, dept, pos, br]); }); return { ok:true, count: staff.length }; } function getStaff() { const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(SHEET_STAFF); if(!sheet || sheet.getLastRow()<2) return { ok:true, staff:[] }; var lastRow = sheet.getLastRow(); if(lastRow < 2) return { ok:true, staff:[] }; var rows = sheet.getRange(2,1,lastRow-1,5).getValues(); var staff = rows.filter(function(r){ return String(r[1]||'').trim(); }).map(function(r){ return { id:String(r[0]||''), name:String(r[1]||''), dept:String(r[2]||''), position:String(r[3]||''), branch:String(r[4]||'') }; }); return { ok:true, staff:staff }; } // ============================================================ // CHECK NAME — ตรวจว่าชื่อนี้เคยทำไปแล้วหรือยัง // ============================================================ function checkName(name) { if (!name) return { exists: false }; const ss = SpreadsheetApp.openById(SHEET_ID); const rSheet = ss.getSheetByName(SHEET_RESPONSES); if (!rSheet || rSheet.getLastRow() < 2) return { exists: false }; const names = rSheet.getRange(2, 2, rSheet.getLastRow()-1, 1).getValues() .map(function(r){ return String(r[0]).trim().toLowerCase(); }); const exists = names.indexOf(String(name).trim().toLowerCase()) > -1; return { exists: exists }; } // ============================================================ // HELPER // ============================================================ function styleHeader(sheet) { try { sheet.getRange(1, 1, 1, sheet.getLastColumn()) .setFontWeight('bold').setBackground('#1a56db').setFontColor('white'); } catch(e){} }
